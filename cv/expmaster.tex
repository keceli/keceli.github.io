\ressubheading{}{}{\bf Computational atomic physics at Bilkent} {2004 - 2006}
\begin{itemize}
\resitem{Demonstrated that a simple analytical ansatz, which has been used to describe the intensity profile of the similariton laser, can be utilized as a variational wave function to solve the Gross-Pitaevskii equation for a wide range of interaction parameters.}
\resitem{Derived dispersion relations for collective modes of a vortex lattice formed in a Bose-Einstein condensate, numerically calculated elastic constants for different lattice geometries and identified structural phase transitions.}
\end{itemize}
